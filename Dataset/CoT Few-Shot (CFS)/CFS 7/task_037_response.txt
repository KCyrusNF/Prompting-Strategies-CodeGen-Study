def reverse_string(s):
    # Initialize an empty string to hold the reversed characters
    reversed_str = ""
    
    # Loop through the string from the last character to the first
    for i in range(len(s)-1, -1, -1):
        reversed_str += s[i]
    
    return reversed_str
