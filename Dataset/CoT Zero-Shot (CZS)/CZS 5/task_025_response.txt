def max_of_three(a, b, c):
    maximum = a  # Step 1: Assume maximum is a
    if b > maximum:  # Step 2: If b is greater, update the maximum
        maximum = b
    if c > maximum:  # Step 3: If c is greater, update the maximum
        maximum = c
    return maximum  # Return the final maximum value
